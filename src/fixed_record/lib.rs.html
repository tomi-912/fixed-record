1//! Fixed-width record helpers generated from plain Rust structs.
2//!
3//! `fixed-record` lets application code describe a fixed-width record once as a Rust struct,
4//! then use generated builders, parsers, field accessors, stream readers/writers, and searchable
5//! list APIs.
6//!
7//! `fixed-record` は、固定長レコードのレイアウトを Rust の struct として一度だけ定義し、
8//! builder、parser、フィールド accessor、stream 用 Reader/Writer、検索可能な List API を
9//! 生成して使うための crate です。
10//!
11//! Most users should depend only on this crate and import [`prelude`].
12//! The `fixed-record-macros` crate is the proc-macro implementation detail re-exported here.
13//!
14//! 通常の利用者はこの crate だけに依存し、[`prelude`] を import すれば十分です。
15//! `fixed-record-macros` はここから再エクスポートされる proc macro 実装用 crate です。
16//!
17//! # Installation / インストール
18//!
19//! Before the crate is published to crates.io, depend on the Git repository.
20//!
21//! crates.io に公開する前は、Git repository 依存として追加します。
22//!
23//! ```toml
24//! [dependencies]
25//! fixed-record = { git = "https://github.com/tomi-912/fixed-record.git" }
26//! ```
27//!
28//! After publishing to crates.io, use the versioned dependency.
29//!
30//! crates.io 公開後は、version 指定の依存として追加できます。
31//!
32//! ```toml
33//! [dependencies]
34//! fixed-record = "0.1"
35//! ```
36//!
37//! # Define a Record / レコードを定義する
38//!
39//! Add [`macro@fixed_record`] to a struct whose fields are [`Fixed<N>`](Fixed). Each field width is
40//! a byte width, not a character count.
41//!
42//! [`macro@fixed_record`] を [`Fixed<N>`](Fixed) フィールドだけを持つ struct に付けます。
43//! 各フィールド幅は文字数ではなくバイト数です。
44//!
45//! ```
46//! use fixed_record::prelude::*;
47//!
48//! #[fixed_record]
49//! struct Order {
50//!     customer_id: Fixed<4>,
51//!     order_no: Fixed<6>,
52//!     amount: Fixed<8>,
53//! }
54//!
55//! assert_eq!(Order::TOTAL_LEN, 18);
56//! assert_eq!(Order::FIELD_SIZE_CUSTOMER_ID, 4);
57//! assert_eq!(Order::offset_of(OrderField::Amount), 10);
58//! assert_eq!(Order::name_of(OrderField::OrderNo), "order_no");
59//! assert_eq!(OrderField::Amount.as_str(), "amount");
60//! assert_eq!(OrderField::Amount.size(), 8);
61//! ```
62//!
63//! Generated names follow the input struct:
64//!
65//! 生成名は入力 struct から決まります。
66//! `OrderField` and `OrderList` use the same visibility as `Order`.
67//! `OrderField` と `OrderList` は `Order` と同じ可視性で生成されます。
68//!
69//! - `Order`: the record type / レコード型
70//! - `OrderField`: the field enum / フィールド enum
71//! - `OrderList`: the searchable list type when the default `list` feature is enabled /
72//!   default feature の `list` が有効な場合の検索用 list 型
73//!
74//! # Build Records / レコードを作成する
75//!
76//! `builder()` starts with `CLEAR_BYTE` padding. Plain `with_*` methods write from the beginning of
77//! the field, while numeric helpers write zero-padded strings.
78//!
79//! `builder()` は `CLEAR_BYTE` で埋めた値から開始します。通常の `with_*` はフィールド先頭から
80//! 上書きし、数値用 helper はゼロ埋め文字列を書き込みます。
81//!
82//! ```
83//! use fixed_record::prelude::*;
84//!
85//! #[fixed_record]
86//! struct Order {
87//!     customer_id: Fixed<4>,
88//!     order_no: Fixed<6>,
89//!     amount: Fixed<8>,
90//! }
91//!
92//! let order = Order::builder()
93//!     .with_customer_id("C001")
94//!     .with_order_no("A00042")
95//!     .with_amount_int(1250)
96//!     .build();
97//!
98//! assert_eq!(order.customer_id(), b"C001");
99//! assert_eq!(order.order_no_str().unwrap(), "A00042");
100//! assert_eq!(order.amount(), b"00001250");
101//! assert_eq!(order.byte_len(), Order::TOTAL_LEN);
102//! ```
103//!
104//! Use `try_with_*_int` or `try_with_*_int_signed` when overflow must be an error. The non-`try`
105//! variants keep the leading bytes when a formatted number is too wide.
106//!
107//! 幅超過をエラーとして扱いたい場合は `try_with_*_int` / `try_with_*_int_signed` を使います。
108//! `try` なしの variant は、幅を超えた数値の先頭側を残します。
109//!
110//! ```
111//! use fixed_record::prelude::*;
112//!
113//! #[fixed_record]
114//! struct Balance {
115//!     account: Fixed<4>,
116//!     signed_amount: Fixed<6>,
117//! }
118//!
119//! let balance = Balance::builder()
120//!     .with_account("A001")
121//!     .try_with_signed_amount_int_signed(-42)
122//!     .unwrap()
123//!     .build();
124//!
125//! assert_eq!(balance.signed_amount(), b"-00042");
126//!
127//! let err = Balance::builder()
128//!     .try_with_signed_amount_int_signed(-123456)
129//!     .unwrap_err();
130//! assert!(matches!(err, fixed_record::Error::FieldOverflow { field: "signed_amount", .. }));
131//! ```
132//!
133//! # Parse and Serialize / パースとシリアライズ
134//!
135//! `parse` / `parse_str` read fixed-width bytes into an owned record. `to_bytes` copies the record
136//! back to its fixed-width byte array.
137//!
138//! `parse` / `parse_str` は固定長バイト列を所有値のレコードへ変換します。`to_bytes` はレコードを
139//! 固定長バイト配列へコピーして返します。
140//!
141//! ```
142//! use fixed_record::prelude::*;
143//!
144//! #[fixed_record]
145//! struct Order {
146//!     customer_id: Fixed<4>,
147//!     order_no: Fixed<6>,
148//!     amount: Fixed<8>,
149//! }
150//!
151//! let raw = b"C001A0004200001250";
152//! let order = Order::parse(raw).unwrap();
153//!
154//! assert_eq!(order.get_field_trimmed(OrderField::CustomerId).unwrap(), "C001");
155//! assert_eq!(order.get_field_as::<u32>(OrderField::Amount).unwrap(), 1250);
156//! assert_eq!(order.to_bytes(), *raw);
157//!
158//! let same = Order::parse_str("C001A0004200001250").unwrap();
159//! assert_eq!(same.to_bytes(), *raw);
160//! ```
161//!
162//! # Dynamic Field Operations / 動的フィールド操作
163//!
164//! Field enum values let you write generic code that does not hard-code accessor method names.
165//! `set_field_*` clears the target field first; `set_field_*_no_clear` preserves trailing bytes.
166//!
167//! フィールド enum を使うと、accessor 名を固定せずに汎用的な処理を書けます。
168//! `set_field_*` は対象フィールドを先にクリアし、`set_field_*_no_clear` は後続バイトを残します。
169//!
170//! ```
171//! use fixed_record::prelude::*;
172//!
173//! #[fixed_record]
174//! struct Order {
175//!     customer_id: Fixed<4>,
176//!     order_no: Fixed<6>,
177//!     amount: Fixed<8>,
178//! }
179//!
180//! let mut order = Order::spaced()
181//!     .with_customer_id("C001")
182//!     .with_order_no("A00042")
183//!     .with_amount_int(1250)
184//!     .build();
185//!
186//! order.set_field_str(OrderField::OrderNo, "B7");
187//! assert_eq!(order.order_no(), b"B7    ");
188//!
189//! order.set_field_str_no_clear(OrderField::OrderNo, "C8");
190//! assert_eq!(order.order_no(), b"C8    ");
191//!
192//! order.fill_field_zero(OrderField::Amount);
193//! assert_eq!(order.amount(), &[0, 0, 0, 0, 0, 0, 0, 0]);
194//! ```
195//!
196//! # Bulk Application Helpers / 一括流し込み helper
197//!
198//! `apply_bytes` and `apply_str` split input across fields in declaration order. The `_from`
199//! variants begin at a selected field.
200//!
201//! `apply_bytes` / `apply_str` は定義順にフィールド幅ごとに入力を流し込みます。`_from` variant は
202//! 指定フィールドから開始します。
203//!
204//! ```
205//! use fixed_record::prelude::*;
206//!
207//! #[fixed_record]
208//! struct Order {
209//!     customer_id: Fixed<4>,
210//!     order_no: Fixed<6>,
211//!     amount: Fixed<8>,
212//! }
213//!
214//! let order = Order::spaced().apply_bytes(b"C001A0004200001250");
215//! assert_eq!(order.customer_id(), b"C001");
216//! assert_eq!(order.order_no(), b"A00042");
217//! assert_eq!(order.amount(), b"00001250");
218//!
219//! let changed_tail = Order::spaced()
220//!     .with_customer_id("C999")
221//!     .apply_str_from(OrderField::OrderNo, "B0007700000900");
222//!
223//! assert_eq!(changed_tail.customer_id(), b"C999");
224//! assert_eq!(changed_tail.order_no(), b"B00077");
225//! assert_eq!(changed_tail.amount(), b"00000900");
226//! ```
227//!
228//! # Reader and Writer / Reader / Writer
229//!
230//! [`Reader`] reads records sequentially from any `BufRead`. `Reader::new` expects LF (`\n`) after
231//! records. Use [`Reader::with_separator`] when the input uses a different separator or no
232//! separator. When configured, the separator must also appear after the final record. [`Writer`]
233//! writes `to_bytes()` output and appends LF by default.
234//!
235//! [`Reader`] は任意の `BufRead` から固定長レコードを順に読みます。`Reader::new` はレコード後ろに
236//! LF (`\n`) がある入力を想定します。別の区切り、または区切りなしの入力を読む場合は
237//! [`Reader::with_separator`] で指定します。区切りを設定した場合は、最終レコードの後ろにもその区切りが必要です。
238//! [`Writer`] は `to_bytes()` の結果を書き出し、デフォルトでは LF を付けます。
239//!
240//! ```
241//! use fixed_record::prelude::*;
242//! use std::io::{BufReader, Cursor};
243//!
244//! #[fixed_record]
245//! struct Order {
246//!     customer_id: Fixed<4>,
247//!     order_no: Fixed<6>,
248//!     amount: Fixed<8>,
249//! }
250//!
251//! let first = Order::builder()
252//!     .with_customer_id("C001")
253//!     .with_order_no("A00001")
254//!     .with_amount_int(100)
255//!     .build();
256//! let second = Order::builder()
257//!     .with_customer_id("C002")
258//!     .with_order_no("A00002")
259//!     .with_amount_int(200)
260//!     .build();
261//!
262//! let mut output = Vec::new();
263//! let mut writer = Writer::new(&mut output)
264//!     .with_separator(RecordSeparator::Crlf);
265//! writer.write_record(&first).unwrap();
266//! writer.write_record(&second).unwrap();
267//! writer.flush().unwrap();
268//! drop(writer);
269//!
270//! let mut reader = Reader::<_, Order>::new(BufReader::new(Cursor::new(output)))
271//!     .with_separator(RecordSeparator::Crlf);
272//! assert_eq!(reader.next().unwrap().unwrap().order_no(), b"A00001");
273//! assert_eq!(reader.next().unwrap().unwrap().order_no(), b"A00002");
274//! assert!(reader.next().is_none());
275//! ```
276//!
277//! Sequence checks can validate that input is sorted by selected fields.
278//!
279//! シーケンスチェックを使うと、入力が指定フィールド順に昇順であることを検証できます。
280//!
281//! ```
282//! use fixed_record::prelude::*;
283//! use std::io::{BufReader, Cursor};
284//!
285//! #[fixed_record]
286//! struct Order {
287//!     customer_id: Fixed<4>,
288//!     order_no: Fixed<6>,
289//!     amount: Fixed<8>,
290//! }
291//!
292//! let first = Order::builder().with_customer_id("C001").with_order_no("A00001").with_amount_int(100).build();
293//! let second = Order::builder().with_customer_id("C001").with_order_no("A00002").with_amount_int(200).build();
294//!
295//! let mut input = Vec::new();
296//! input.extend_from_slice(&first.to_bytes());
297//! input.push(b'\n');
298//! input.extend_from_slice(&second.to_bytes());
299//! input.push(b'\n');
300//!
301//! let mut reader = Reader::<_, Order>::new(BufReader::new(Cursor::new(input)))
302//!     .with_sequence_check([OrderField::CustomerId, OrderField::OrderNo]);
303//!
304//! assert!(reader.next().unwrap().is_ok());
305//! assert!(reader.next().unwrap().is_ok());
306//! assert!(reader.next().is_none());
307//! ```
308//!
309//! # Searchable Lists / 検索可能な List
310//!
311//! With the default `list` feature, the macro generates `{StructName}List`. It stores records,
312//! maintains field indexes, and supports lookup, update, logical removal, vacuuming, sorting, exact
313//! searches, padded searches, prefix searches, and range searches.
314//!
315//! default feature の `list` が有効な場合、macro は `{StructName}List` を生成します。レコードを
316//! 保持し、フィールド index を管理し、lookup、update、論理削除、vacuum、sort、完全一致検索、
317//! padding を考慮した検索、prefix 検索、range 検索を提供します。
318//!
319//! ```
320//! use fixed_record::prelude::*;
321//!
322//! #[fixed_record]
323//! struct Order {
324//!     customer_id: Fixed<4>,
325//!     order_no: Fixed<6>,
326//!     amount: Fixed<8>,
327//! }
328//!
329//! let mut list = OrderList::new();
330//! let id_first = list.insert(Order::builder()
331//!     .with_customer_id("C001")
332//!     .with_order_no("A00002")
333//!     .with_amount_int(200)
334//!     .build());
335//! let id_second = list.insert(Order::builder()
336//!     .with_customer_id("C001")
337//!     .with_order_no("A00001")
338//!     .with_amount_int(100)
339//!     .build());
340//!
341//! assert_eq!(list.len(), 2);
342//! assert_eq!(list.get(id_first).unwrap().amount(), b"00000200");
343//!
344//! let exact = list.find_by(OrderField::CustomerId, *b"C001");
345//! assert_eq!(exact.len(), 2);
346//!
347//! let padded = list.try_find_by(OrderField::OrderNo, b"A00001").unwrap();
348//! assert_eq!(padded[0].amount(), b"00000100");
349//!
350//! let prefix = list.try_find_by_prefix(OrderField::OrderNo, b"A000").unwrap();
351//! assert_eq!(prefix.len(), 2);
352//!
353//! let first_by_order_no = list.try_first_sorted_by(OrderField::OrderNo).unwrap();
354//! assert_eq!(first_by_order_no.order_no(), b"A00001");
355//!
356//! list.update(id_first, Order::builder()
357//!     .with_customer_id("C002")
358//!     .with_order_no("B00001")
359//!     .with_amount_int(300)
360//!     .build());
361//! assert!(list.try_find_by(OrderField::CustomerId, b"C001").unwrap().len() == 1);
362//!
363//! assert!(list.remove(id_second));
364//! assert_eq!(list.len(), 1);
365//! assert_eq!(list.all_ids().len(), 2);
366//! list.vacuum();
367//! assert_eq!(list.all_ids(), vec![id_first]);
368//! ```
369//!
370//! Range searches use [`Fixed<N>`](Fixed) bounds whose width matches the searched field.
371//!
372//! range 検索では、検索対象フィールドと同じ幅の [`Fixed<N>`](Fixed) 境界値を使います。
373//!
374//! ```
375//! use fixed_record::prelude::*;
376//!
377//! #[fixed_record]
378//! struct Order {
379//!     customer_id: Fixed<4>,
380//!     order_no: Fixed<6>,
381//!     amount: Fixed<8>,
382//! }
383//!
384//! let mut list = OrderList::new();
385//! list.insert(Order::builder().with_customer_id("C001").with_order_no("A00001").with_amount_int(100).build());
386//! list.insert(Order::builder().with_customer_id("C001").with_order_no("A00002").with_amount_int(200).build());
387//! list.insert(Order::builder().with_customer_id("C001").with_order_no("A00003").with_amount_int(300).build());
388//!
389//! let low = Fixed::<8>::from(*b"00000150");
390//! let high = Fixed::<8>::from(*b"00000300");
391//! let found = list.find_range_by(OrderField::Amount, low..=high);
392//!
393//! assert_eq!(found.len(), 2);
394//! assert_eq!(found[0].amount(), b"00000200");
395//! assert_eq!(found[1].amount(), b"00000300");
396//! ```
397//!
398//! # Clear Byte and Initialization / clear_byte と初期化
399//!
400//! By default, [`macro@fixed_record`] uses spaces (`0x20`) as `CLEAR_BYTE`. You can change this with
401//! `clear_byte = ZERO`, `clear_byte = SPACE`, a byte literal, or an integer from `0` to `255`.
402//!
403//! 標準では [`macro@fixed_record`] は半角スペース (`0x20`) を `CLEAR_BYTE` として使います。
404//! `clear_byte = ZERO`、`clear_byte = SPACE`、byte literal、`0` から `255` の整数で変更できます。
405//!
406//! ```
407//! use fixed_record::prelude::*;
408//!
409//! #[fixed_record(clear_byte = ZERO)]
410//! struct ZeroPaddedName {
411//!     name: Fixed<6>,
412//! }
413//!
414//! let mut record = ZeroPaddedName::builder().with_name("Alice").build();
415//! record.set_field_str(ZeroPaddedNameField::Name, "Bo");
416//!
417//! assert_eq!(ZeroPaddedName::CLEAR_BYTE, 0x00);
418//! assert_eq!(record.name(), b"Bo\0\0\0\0");
419//! assert_eq!(ZeroPaddedName::zeroed().name(), b"\0\0\0\0\0\0");
420//! assert_eq!(ZeroPaddedName::spaced().name(), b"      ");
421//! assert_eq!(ZeroPaddedName::cleared().name(), b"\0\0\0\0\0\0");
422//! ```
423//!
424//! # FixedRecord Trait / FixedRecord trait
425//!
426//! Generated records implement [`FixedRecord`], so generic code can parse or serialize any generated
427//! record type accepted by [`Reader`] and [`Writer`].
428//!
429//! 生成されたレコードは [`FixedRecord`] を実装します。そのため [`Reader`] / [`Writer`] が扱える
430//! 任意の生成レコードに対して generic な parse / serialize 処理を書けます。
431//!
432//! ```
433//! use fixed_record::prelude::*;
434//!
435//! #[fixed_record]
436//! struct Order {
437//!     customer_id: Fixed<4>,
438//!     order_no: Fixed<6>,
439//!     amount: Fixed<8>,
440//! }
441//!
442//! fn serialize<T: FixedRecord>(record: &T) -> Vec<u8> {
443//!     record.to_bytes()
444//! }
445//!
446//! let order = Order::builder()
447//!     .with_customer_id("C001")
448//!     .with_order_no("A00042")
449//!     .with_amount_int(1250)
450//!     .build();
451//!
452//! assert_eq!(serialize(&order), b"C001A0004200001250");
453//! ```
454//!

469//! # Zerocopy Support / zerocopy 対応
470//!
471//! Generated records always use `#[repr(C)]` and derive `zerocopy::FromBytes`,
472//! `zerocopy::IntoBytes`, `zerocopy::Immutable`, and `zerocopy::KnownLayout`.
473//! The zerocopy traits are re-exported from [`prelude`].
474//!
475//! 生成レコードには常に `#[repr(C)]` が付き、`zerocopy::FromBytes`、
476//! `zerocopy::IntoBytes`、`zerocopy::Immutable`、`zerocopy::KnownLayout` を derive します。
477//! zerocopy traits は [`prelude`] から再エクスポートされます。
478//!
479//! ```
480//! use fixed_record::prelude::*;
481//!
482//! #[fixed_record]
483//! struct Order {
484//!     customer_id: Fixed<4>,
485//!     order_no: Fixed<6>,
486//!     amount: Fixed<8>,
487//! }
488//!
489//! let raw = b"C001A0004200001250";
490//!
491//! let order = Order::ref_from_bytes(raw).unwrap();
492//! assert_eq!(order.customer_id(), b"C001");
493//! assert_eq!(order.as_bytes(), raw);
494//! assert_eq!(order.as_str().unwrap(), "C001A0004200001250");
495//! ```
496//!
497//! Use `ref_from_bytes_prefix` when the input may contain trailing bytes after the first record.
498//!
499//! 先頭1レコードの後ろに余りバイトが続く可能性がある場合は、`ref_from_bytes_prefix` を使います。
500//!
501//! ```
502//! use fixed_record::prelude::*;
503//!
504//! #[fixed_record]
505//! struct Order {
506//!     customer_id: Fixed<4>,
507//!     order_no: Fixed<6>,
508//!     amount: Fixed<8>,
509//! }
510//!
511//! let raw = b"C001A0004200001250rest";
512//! let order = Order::ref_from_bytes_prefix(raw).unwrap();
513//! assert_eq!(order.customer_id(), b"C001");
514//! ```
515//!
516//! String inputs can use `ref_from_str` or `ref_from_str_prefix`.
517//!
518//! 文字列入力では `ref_from_str` または `ref_from_str_prefix` を使えます。
519//!
520//! ```
521//! use fixed_record::prelude::*;
522//!
523//! #[fixed_record]
524//! struct Order {
525//!     customer_id: Fixed<4>,
526//!     order_no: Fixed<6>,
527//!     amount: Fixed<8>,
528//! }
529//!
530//! let order = Order::ref_from_str_prefix("C001A0004200001250rest").unwrap();
531//! assert_eq!(order.order_no_str().unwrap(), "A00042");
532//! ```
