1//! Fixed-width record helpers generated from plain Rust structs.
2//!
3//! `fixed-record` lets application code describe a fixed-width record once as a Rust struct,
4//! then use generated builders, parsers, field accessors, stream readers/writers, and searchable
5//! list APIs.
6//!
7//! `fixed-record` は、固定長レコードのレイアウトを Rust の struct として一度だけ定義し、
8//! builder、parser、フィールド accessor、stream 用 Reader/Writer、検索可能な List API を
9//! 生成して使うための crate です。
10//!
11//! Most users should depend only on this crate and import [`prelude`].
12//! The `fixed-record-macros` crate is the proc-macro implementation detail re-exported here.
13//!
14//! 通常の利用者はこの crate だけに依存し、[`prelude`] を import すれば十分です。
15//! `fixed-record-macros` はここから再エクスポートされる proc macro 実装用 crate です。
16//!
17//! # Installation / インストール
18//!
19//! Before the crate is published to crates.io, depend on the Git repository.
20//!
21//! crates.io に公開する前は、Git repository 依存として追加します。
22//!
23//! ```toml
24//! [dependencies]
25//! fixed-record = { git = "https://github.com/tomi-912/fixed-record.git" }
26//! ```
27//!
28//! After publishing to crates.io, use the versioned dependency.
29//!
30//! crates.io 公開後は、version 指定の依存として追加できます。
31//!
32//! ```toml
33//! [dependencies]
34//! fixed-record = "0.1"
35//! ```
36//!
37//! # Define a Record / レコードを定義する
38//!
39//! Add [`macro@fixed_record`] to a struct whose fields are [`Fixed<N>`](Fixed). Each field width is
40//! a byte width, not a character count.
41//!
42//! [`macro@fixed_record`] を [`Fixed<N>`](Fixed) フィールドだけを持つ struct に付けます。
43//! 各フィールド幅は文字数ではなくバイト数です。
44//!
45//! ```
46//! use fixed_record::prelude::*;
47//!
48//! #[fixed_record]
49//! pub struct Order {
50//!     pub customer_id: Fixed<4>,
51//!     pub order_no: Fixed<6>,
52//!     pub amount: Fixed<8>,
53//! }
54//!
55//! assert_eq!(Order::TOTAL_LEN, 18);
56//! assert_eq!(Order::FIELD_SIZE_CUSTOMER_ID, 4);
57//! assert_eq!(Order::offset_of(OrderField::Amount), 10);
58//! assert_eq!(Order::name_of(OrderField::OrderNo), "order_no");
59//! assert_eq!(OrderField::Amount.as_str(), "amount");
60//! assert_eq!(OrderField::Amount.size(), 8);
61//! ```
62//!
63//! Generated names follow the input struct:
64//!
65//! 生成名は入力 struct から決まります。
66//!
67//! - `Order`: the record type / レコード型
68//! - `OrderField`: the field enum / フィールド enum
69//! - `OrderList`: the searchable list type when the default `list` feature is enabled /
70//!   default feature の `list` が有効な場合の検索用 list 型
71//!
72//! # Build Records / レコードを作成する
73//!
74//! `builder()` starts with `CLEAR_BYTE` padding. Plain `with_*` methods write from the beginning of
75//! the field, while numeric helpers write zero-padded strings.
76//!
77//! `builder()` は `CLEAR_BYTE` で埋めた値から開始します。通常の `with_*` はフィールド先頭から
78//! 上書きし、数値用 helper はゼロ埋め文字列を書き込みます。
79//!
80//! ```
81//! use fixed_record::prelude::*;
82//!
83//! #[fixed_record]
84//! pub struct Order {
85//!     customer_id: Fixed<4>,
86//!     order_no: Fixed<6>,
87//!     amount: Fixed<8>,
88//! }
89//!
90//! let order = Order::builder()
91//!     .with_customer_id("C001")
92//!     .with_order_no("A00042")
93//!     .with_amount_int(1250)
94//!     .build();
95//!
96//! assert_eq!(order.customer_id(), b"C001");
97//! assert_eq!(order.order_no_str().unwrap(), "A00042");
98//! assert_eq!(order.amount(), b"00001250");
99//! assert_eq!(order.byte_len(), Order::TOTAL_LEN);
100//! ```
101//!
102//! Use `try_with_*_int` or `try_with_*_int_signed` when overflow must be an error. The non-`try`
103//! variants keep the leading bytes when a formatted number is too wide.
104//!
105//! 幅超過をエラーとして扱いたい場合は `try_with_*_int` / `try_with_*_int_signed` を使います。
106//! `try` なしの variant は、幅を超えた数値の先頭側を残します。
107//!
108//! ```
109//! use fixed_record::prelude::*;
110//!
111//! #[fixed_record]
112//! pub struct Balance {
113//!     account: Fixed<4>,
114//!     signed_amount: Fixed<6>,
115//! }
116//!
117//! let balance = Balance::builder()
118//!     .with_account("A001")
119//!     .try_with_signed_amount_int_signed(-42)
120//!     .unwrap()
121//!     .build();
122//!
123//! assert_eq!(balance.signed_amount(), b"-00042");
124//!
125//! let err = Balance::builder()
126//!     .try_with_signed_amount_int_signed(-123456)
127//!     .unwrap_err();
128//! assert!(matches!(err, fixed_record::Error::FieldOverflow { field: "signed_amount", .. }));
129//! ```
130//!
131//! # Parse and Serialize / パースとシリアライズ
132//!
133//! `parse` / `parse_str` read fixed-width bytes into an owned record. `to_bytes` copies the record
134//! back to its fixed-width byte array.
135//!
136//! `parse` / `parse_str` は固定長バイト列を所有値のレコードへ変換します。`to_bytes` はレコードを
137//! 固定長バイト配列へコピーして返します。
138//!
139//! ```
140//! use fixed_record::prelude::*;
141//!
142//! #[fixed_record]
143//! pub struct Order {
144//!     customer_id: Fixed<4>,
145//!     order_no: Fixed<6>,
146//!     amount: Fixed<8>,
147//! }
148//!
149//! let raw = b"C001A0004200001250";
150//! let order = Order::parse(raw).unwrap();
151//!
152//! assert_eq!(order.get_field_trimmed(OrderField::CustomerId).unwrap(), "C001");
153//! assert_eq!(order.get_field_as::<u32>(OrderField::Amount).unwrap(), 1250);
154//! assert_eq!(order.to_bytes(), *raw);
155//!
156//! let same = Order::parse_str("C001A0004200001250").unwrap();
157//! assert_eq!(same.to_bytes(), *raw);
158//! ```
159//!
160//! # Dynamic Field Operations / 動的フィールド操作
161//!
162//! Field enum values let you write generic code that does not hard-code accessor method names.
163//! `set_field_*` clears the target field first; `set_field_*_no_clear` preserves trailing bytes.
164//!
165//! フィールド enum を使うと、accessor 名を固定せずに汎用的な処理を書けます。
166//! `set_field_*` は対象フィールドを先にクリアし、`set_field_*_no_clear` は後続バイトを残します。
167//!
168//! ```
169//! use fixed_record::prelude::*;
170//!
171//! #[fixed_record]
172//! pub struct Order {
173//!     customer_id: Fixed<4>,
174//!     order_no: Fixed<6>,
175//!     amount: Fixed<8>,
176//! }
177//!
178//! let mut order = Order::spaced()
179//!     .with_customer_id("C001")
180//!     .with_order_no("A00042")
181//!     .with_amount_int(1250)
182//!     .build();
183//!
184//! order.set_field_str(OrderField::OrderNo, "B7");
185//! assert_eq!(order.order_no(), b"B7    ");
186//!
187//! order.set_field_str_no_clear(OrderField::OrderNo, "C8");
188//! assert_eq!(order.order_no(), b"C8    ");
189//!
190//! order.fill_field_zero(OrderField::Amount);
191//! assert_eq!(order.amount(), &[0, 0, 0, 0, 0, 0, 0, 0]);
192//! ```
193//!
194//! # Bulk Application Helpers / 一括流し込み helper
195//!
196//! `apply_bytes` and `apply_str` split input across fields in declaration order. The `_from`
197//! variants begin at a selected field.
198//!
199//! `apply_bytes` / `apply_str` は定義順にフィールド幅ごとに入力を流し込みます。`_from` variant は
200//! 指定フィールドから開始します。
201//!
202//! ```
203//! use fixed_record::prelude::*;
204//!
205//! #[fixed_record]
206//! pub struct Order {
207//!     customer_id: Fixed<4>,
208//!     order_no: Fixed<6>,
209//!     amount: Fixed<8>,
210//! }
211//!
212//! let order = Order::spaced().apply_bytes(b"C001A0004200001250");
213//! assert_eq!(order.customer_id(), b"C001");
214//! assert_eq!(order.order_no(), b"A00042");
215//! assert_eq!(order.amount(), b"00001250");
216//!
217//! let changed_tail = Order::spaced()
218//!     .with_customer_id("C999")
219//!     .apply_str_from(OrderField::OrderNo, "B0007700000900");
220//!
221//! assert_eq!(changed_tail.customer_id(), b"C999");
222//! assert_eq!(changed_tail.order_no(), b"B00077");
223//! assert_eq!(changed_tail.amount(), b"00000900");
224//! ```
225//!
226//! # Reader and Writer / Reader / Writer
227//!
228//! [`Reader`] reads records sequentially from any `BufRead`. A separator immediately after each
229//! record (`\n`, `\r`, `\r\n`, or `,`) is skipped automatically. [`Writer`] writes `to_bytes()`
230//! output and appends a [`RecordSeparator`].
231//!
232//! [`Reader`] は任意の `BufRead` から固定長レコードを順に読みます。各レコード直後の区切り
233//! (`\n`, `\r`, `\r\n`, `,`) は自動的に読み飛ばします。[`Writer`] は `to_bytes()` の結果を
234//! 書き出し、[`RecordSeparator`] を付けます。
235//!
236//! ```
237//! use fixed_record::prelude::*;
238//! use std::io::{BufReader, Cursor};
239//!
240//! #[fixed_record]
241//! pub struct Order {
242//!     customer_id: Fixed<4>,
243//!     order_no: Fixed<6>,
244//!     amount: Fixed<8>,
245//! }
246//!
247//! let first = Order::builder()
248//!     .with_customer_id("C001")
249//!     .with_order_no("A00001")
250//!     .with_amount_int(100)
251//!     .build();
252//! let second = Order::builder()
253//!     .with_customer_id("C002")
254//!     .with_order_no("A00002")
255//!     .with_amount_int(200)
256//!     .build();
257//!
258//! let mut output = Vec::new();
259//! let mut writer = Writer::new(&mut output)
260//!     .with_separator(RecordSeparator::Crlf);
261//! writer.write_record(&first).unwrap();
262//! writer.write_record(&second).unwrap();
263//! writer.flush().unwrap();
264//! drop(writer);
265//!
266//! let mut reader = Reader::<_, Order>::new(BufReader::new(Cursor::new(output)));
267//! assert_eq!(reader.next().unwrap().unwrap().order_no(), b"A00001");
268//! assert_eq!(reader.next().unwrap().unwrap().order_no(), b"A00002");
269//! assert!(reader.next().is_none());
270//! ```
271//!
272//! Sequence checks can validate that input is sorted by selected fields.
273//!
274//! シーケンスチェックを使うと、入力が指定フィールド順に昇順であることを検証できます。
275//!
276//! ```
277//! use fixed_record::prelude::*;
278//! use std::io::{BufReader, Cursor};
279//!
280//! #[fixed_record]
281//! pub struct Order {
282//!     customer_id: Fixed<4>,
283//!     order_no: Fixed<6>,
284//!     amount: Fixed<8>,
285//! }
286//!
287//! let first = Order::builder().with_customer_id("C001").with_order_no("A00001").with_amount_int(100).build();
288//! let second = Order::builder().with_customer_id("C001").with_order_no("A00002").with_amount_int(200).build();
289//!
290//! let mut input = Vec::new();
291//! input.extend_from_slice(&first.to_bytes());
292//! input.push(b'\n');
293//! input.extend_from_slice(&second.to_bytes());
294//! input.push(b'\n');
295//!
296//! let mut reader = Reader::<_, Order>::new(BufReader::new(Cursor::new(input)))
297//!     .with_sequence_check([OrderField::CustomerId, OrderField::OrderNo]);
298//!
299//! assert!(reader.next().unwrap().is_ok());
300//! assert!(reader.next().unwrap().is_ok());
301//! assert!(reader.next().is_none());
302//! ```
303//!
304//! # Searchable Lists / 検索可能な List
305//!
306//! With the default `list` feature, the macro generates `{StructName}List`. It stores records,
307//! maintains field indexes, and supports lookup, update, logical removal, vacuuming, sorting, exact
308//! searches, padded searches, prefix searches, and range searches.
309//!
310//! default feature の `list` が有効な場合、macro は `{StructName}List` を生成します。レコードを
311//! 保持し、フィールド index を管理し、lookup、update、論理削除、vacuum、sort、完全一致検索、
312//! padding を考慮した検索、prefix 検索、range 検索を提供します。
313//!
314//! ```
315//! use fixed_record::prelude::*;
316//!
317//! #[fixed_record]
318//! pub struct Order {
319//!     customer_id: Fixed<4>,
320//!     order_no: Fixed<6>,
321//!     amount: Fixed<8>,
322//! }
323//!
324//! let mut list = OrderList::new();
325//! let id_first = list.insert(Order::builder()
326//!     .with_customer_id("C001")
327//!     .with_order_no("A00002")
328//!     .with_amount_int(200)
329//!     .build());
330//! let id_second = list.insert(Order::builder()
331//!     .with_customer_id("C001")
332//!     .with_order_no("A00001")
333//!     .with_amount_int(100)
334//!     .build());
335//!
336//! assert_eq!(list.len(), 2);
337//! assert_eq!(list.get(id_first).unwrap().amount(), b"00000200");
338//!
339//! let exact = list.find_by(OrderField::CustomerId, *b"C001");
340//! assert_eq!(exact.len(), 2);
341//!
342//! let padded = list.try_find_by(OrderField::OrderNo, b"A00001").unwrap();
343//! assert_eq!(padded[0].amount(), b"00000100");
344//!
345//! let prefix = list.try_find_by_prefix(OrderField::OrderNo, b"A000").unwrap();
346//! assert_eq!(prefix.len(), 2);
347//!
348//! let first_by_order_no = list.try_first_sorted_by(OrderField::OrderNo).unwrap();
349//! assert_eq!(first_by_order_no.order_no(), b"A00001");
350//!
351//! list.update(id_first, Order::builder()
352//!     .with_customer_id("C002")
353//!     .with_order_no("B00001")
354//!     .with_amount_int(300)
355//!     .build());
356//! assert!(list.try_find_by(OrderField::CustomerId, b"C001").unwrap().len() == 1);
357//!
358//! assert!(list.remove(id_second));
359//! assert_eq!(list.len(), 1);
360//! assert_eq!(list.all_ids().len(), 2);
361//! list.vacuum();
362//! assert_eq!(list.all_ids(), vec![id_first]);
363//! ```
364//!
365//! Range searches use [`Fixed<N>`](Fixed) bounds whose width matches the searched field.
366//!
367//! range 検索では、検索対象フィールドと同じ幅の [`Fixed<N>`](Fixed) 境界値を使います。
368//!
369//! ```
370//! use fixed_record::prelude::*;
371//!
372//! #[fixed_record]
373//! pub struct Order {
374//!     customer_id: Fixed<4>,
375//!     order_no: Fixed<6>,
376//!     amount: Fixed<8>,
377//! }
378//!
379//! let mut list = OrderList::new();
380//! list.insert(Order::builder().with_customer_id("C001").with_order_no("A00001").with_amount_int(100).build());
381//! list.insert(Order::builder().with_customer_id("C001").with_order_no("A00002").with_amount_int(200).build());
382//! list.insert(Order::builder().with_customer_id("C001").with_order_no("A00003").with_amount_int(300).build());
383//!
384//! let low = Fixed::<8>::from(*b"00000150");
385//! let high = Fixed::<8>::from(*b"00000300");
386//! let found = list.find_range_by(OrderField::Amount, low..=high);
387//!
388//! assert_eq!(found.len(), 2);
389//! assert_eq!(found[0].amount(), b"00000200");
390//! assert_eq!(found[1].amount(), b"00000300");
391//! ```
392//!
393//! # Clear Byte and Initialization / clear_byte と初期化
394//!
395//! By default, [`macro@fixed_record`] uses spaces (`0x20`) as `CLEAR_BYTE`. You can change this with
396//! `clear_byte = ZERO`, `clear_byte = SPACE`, a byte literal, or an integer from `0` to `255`.
397//!
398//! 標準では [`macro@fixed_record`] は半角スペース (`0x20`) を `CLEAR_BYTE` として使います。
399//! `clear_byte = ZERO`、`clear_byte = SPACE`、byte literal、`0` から `255` の整数で変更できます。
400//!
401//! ```
402//! use fixed_record::prelude::*;
403//!
404//! #[fixed_record(clear_byte = ZERO)]
405//! pub struct ZeroPaddedName {
406//!     name: Fixed<6>,
407//! }
408//!
409//! let mut record = ZeroPaddedName::builder().with_name("Alice").build();
410//! record.set_field_str(ZeroPaddedNameField::Name, "Bo");
411//!
412//! assert_eq!(ZeroPaddedName::CLEAR_BYTE, 0x00);
413//! assert_eq!(record.name(), b"Bo\0\0\0\0");
414//! assert_eq!(ZeroPaddedName::zeroed().name(), b"\0\0\0\0\0\0");
415//! assert_eq!(ZeroPaddedName::spaced().name(), b"      ");
416//! assert_eq!(ZeroPaddedName::cleared().name(), b"\0\0\0\0\0\0");
417//! ```
418//!
419//! # FixedRecord Trait / FixedRecord trait
420//!
421//! Generated records implement [`FixedRecord`], so generic code can parse or serialize any generated
422//! record type accepted by [`Reader`] and [`Writer`].
423//!
424//! 生成されたレコードは [`FixedRecord`] を実装します。そのため [`Reader`] / [`Writer`] が扱える
425//! 任意の生成レコードに対して generic な parse / serialize 処理を書けます。
426//!
427//! ```
428//! use fixed_record::prelude::*;
429//!
430//! #[fixed_record]
431//! pub struct Order {
432//!     customer_id: Fixed<4>,
433//!     order_no: Fixed<6>,
434//!     amount: Fixed<8>,
435//! }
436//!
437//! fn serialize<T: FixedRecord>(record: &T) -> Vec<u8> {
438//!     record.to_bytes()
439//! }
440//!
441//! let order = Order::builder()
442//!     .with_customer_id("C001")
443//!     .with_order_no("A00042")
444//!     .with_amount_int(1250)
445//!     .build();
446//!
447//! assert_eq!(serialize(&order), b"C001A0004200001250");
448//! ```

450//! # Feature Flags / feature flag
451//!
452//! - `list`: enabled by default. Generates `{StructName}List`.
453//! - `list`: default で有効です。`{StructName}List` を生成します。
454//! - `unchecked`: generates unsafe zero-copy APIs.
455//! - `unchecked`: unsafe なゼロコピー API を生成します。
456//!
457//! To disable List generation:
458//!
459//! List 生成を無効化する場合:
460//!
461//! ```toml
462//! [dependencies]
463//! fixed-record = { version = "0.1", default-features = false }
464//! ```
465//!
466//! With `unchecked`, generated records also expose `as_bytes_unchecked`, `parse_unchecked`,
467//! `from_bytes_unchecked`, and `from_str_unchecked`. These APIs require callers to guarantee that
468//! the Rust struct memory layout exactly matches the fixed-width byte layout.
469//!
470//! `unchecked` を有効にすると、生成レコードに `as_bytes_unchecked`、`parse_unchecked`、
471//! `from_bytes_unchecked`、`from_str_unchecked` も追加されます。これらは Rust の struct memory
472//! layout が固定長バイト配置と完全に一致することを呼び出し側が保証する必要があります。
473//!
474//! ```ignore
475//! use fixed_record::prelude::*;
476//!
477//! #[fixed_record]
478//! pub struct Order {
479//!     customer_id: Fixed<4>,
480//!     order_no: Fixed<6>,
481//!     amount: Fixed<8>,
482//! }
483//!
484//! let raw = b"C001A0004200001250";
485//!
486//! // Requires the `unchecked` feature and a layout guarantee from the caller.
487//! // `unchecked` feature と、呼び出し側による layout 保証が必要です。
488//! let order = unsafe { Order::parse_unchecked(raw).unwrap() };
489//! let bytes = unsafe { order.as_bytes_unchecked() };
490//! assert_eq!(bytes, raw);
491//! ```
492//!
