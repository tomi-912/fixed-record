6/// Separator written after each record by [`Writer`].
7/// [`Writer`] が各レコードの後ろに書き出す区切りです。
8#[derive(Debug, Clone, Copy, PartialEq, Eq)]
9pub enum RecordSeparator {
10    /// No separator.
11    /// 区切りなしです。
12    None,
13    /// Line feed (`\n`).
14    /// LF (`\n`) です。
15    Lf,
16    /// Carriage return (`\r`).
17    /// CR (`\r`) です。
18    Cr,
19    /// Carriage return and line feed (`\r\n`).
20    /// CRLF (`\r\n`) です。
21    Crlf,
22    /// Comma (`,`).
23    /// カンマ (`,`) です。
24    Comma,
25}
26
27impl RecordSeparator {
28    /// Returns the byte sequence for this separator.
29    /// この区切りを表すバイト列を返します。
30    pub const fn as_bytes(self) -> &'static [u8] {
31        match self {
32            Self::None => b"",
33            Self::Lf => b"\n",
34            Self::Cr => b"\r",
35            Self::Crlf => b"\r\n",
36            Self::Comma => b",",
37        }
38    }
39}

41/// Iterator that reads fixed-width records from a stream.
42/// 固定長レコードをストリームから順に読み込むイテレータです。
43///
44/// `Reader::new` expects LF (`\n`) after records. Use [`Reader::with_separator`] when the input
45/// uses a different separator or no separator.
46/// `Reader::new` はレコード後ろに LF (`\n`) がある入力を想定します。別の区切り、または区切りなしの
47/// 入力を読む場合は [`Reader::with_separator`] で指定します。
48///
49/// When a separator is configured, it must appear after every record, including the final record.
50/// 区切りを設定した場合は、最終レコードを含むすべてのレコード後ろにその区切りが必要です。
51///
52/// # Examples
53///
54/// ```
55/// use fixed_record::prelude::*;
56/// use std::io::{BufReader, Cursor};
57///
58/// #[fixed_record]
59/// pub struct Payment {
60///     id: Fixed<4>,
61///     amount: Fixed<6>,
62/// }
63///
64/// let first = Payment::builder()
65///     .with_id("A001")
66///     .with_amount_int(1200)
67///     .build();
68/// let second = Payment::builder()
69///     .with_id("A002")
70///     .with_amount_int(3400)
71///     .build();
72///
73/// let mut input = Vec::new();
74/// input.extend_from_slice(&first.to_bytes());
75/// input.extend_from_slice(b"\r\n");
76/// input.extend_from_slice(&second.to_bytes());
77/// input.extend_from_slice(b"\r\n");
78///
79/// let mut reader = Reader::<_, Payment>::new(BufReader::new(Cursor::new(input)))
80///     .with_separator(RecordSeparator::Crlf);
81///
82/// assert_eq!(
83///     reader.next().unwrap().unwrap().get_field_trimmed(PaymentField::Id).unwrap(),
84///     "A001"
85/// );
86/// assert_eq!(
87///     reader.next().unwrap().unwrap().get_field_trimmed(PaymentField::Amount).unwrap(),
88///     "003400"
89/// );
90/// assert!(reader.next().is_none());
91/// ```
92pub struct Reader<R, T: FixedRecord> {
93    reader: R,
94    separator: RecordSeparator,
95    sequence_fields: Vec<T::Field>,
96    allow_equal_sequence: bool,
97    previous_sequence_key: Option<Vec<Vec<u8>>>,
98    _marker: PhantomData<T>,
99}
100
101impl<R: BufRead, T: FixedRecord> Reader<R, T> {
102    /// Creates a new reader.
103    /// 新しいリーダーを作成します。
104    pub fn new(reader: R) -> Self {
105        Self {
106            reader,
107            separator: RecordSeparator::Lf,
108            sequence_fields: Vec::new(),
109            allow_equal_sequence: true,
110            previous_sequence_key: None,
111            _marker: PhantomData,
112        }
113    }
114
115    /// Configures the separator expected after records.
116    /// レコードの後ろに期待する区切りを設定します。
117    ///
118    /// `RecordSeparator::None` means records are adjacent with no bytes between them.
119    /// `RecordSeparator::None` は、レコード間に区切りバイトがないことを表します。
120    pub fn with_separator(mut self, separator: RecordSeparator) -> Self {
121        self.separator = separator;
122        self
123    }
124
125    /// Enables ascending sequence checks using the specified fields as the key.
126    /// 指定フィールドをキーにした昇順シーケンスチェックを有効にします。
127    pub fn with_sequence_check<F>(mut self, fields: F) -> Self
128    where
129        F: SequenceFields<T>,
130    {
131        self.set_sequence_check_fields(fields.to_sequence_fields());
132        self
133    }
134
135    /// Configures ascending sequence checks using the specified fields as the key.
136    /// 指定フィールドをキーにした昇順シーケンスチェックを設定します。
137    pub fn with_sequence_check_options<F>(mut self, fields: F, allow_equal: bool) -> Self
138    where
139        F: SequenceFields<T>,
140    {
141        self.set_sequence_check_fields(fields.to_sequence_fields());
142        self.allow_equal_sequence = allow_equal;
143        self
144    }
145
146    /// Stores the fields used for sequence checks.
147    /// シーケンスチェック対象フィールドを設定します。
148    fn set_sequence_check_fields(&mut self, fields: Vec<T::Field>) {
149        for (index, field) in fields.iter().enumerate() {
150            if fields[index + 1..].contains(field) {
151                panic!("duplicate sequence check field `{}`", T::field_name(*field));
152            }
153        }
154
155        self.sequence_fields = fields;
156    }
157
158    /// Verifies that the loaded record is ordered after the previous record.
159    /// 読み込んだレコードが前回レコード以降の順序になっているか確認します。
160    fn check_sequence(&mut self, record: &T) -> Result<(), Error> {
161        if self.sequence_fields.is_empty() {
162            return Ok(());
163        }
164
165        let current_key: Vec<Vec<u8>> = self
166            .sequence_fields
167            .iter()
168            .map(|field| record.field_bytes(*field).to_vec())
169            .collect();
170
171        if let Some(previous_key) = &self.previous_sequence_key {
172            let ordering = current_key.cmp(previous_key);
173            let is_error = ordering == Ordering::Less
174                || (ordering == Ordering::Equal && !self.allow_equal_sequence);
175            if is_error {
176                return Err(Error::SequenceError {
177                    fields: self
178                        .sequence_fields
179                        .iter()
180                        .map(|field| T::field_name(*field))
181                        .collect(),
182                    previous: previous_key.clone(),
183                    current: current_key,
184                });
185            }
186        }
187
188        self.previous_sequence_key = Some(current_key);
189        Ok(())
190    }
191}

193impl<R: BufRead, T: FixedRecord> Iterator for Reader<R, T> {
194    type Item = Result<T, Error>;
195
196    /// Reads the next fixed-width record, returning end-of-stream or an error when appropriate.
197    /// 次の固定長レコードを読み込み、終端またはエラーを返します。
198    fn next(&mut self) -> Option<Self::Item> {
199        let mut buf = vec![0u8; T::TOTAL_LEN];
200        let mut read_len = 0;
201
202        while read_len < T::TOTAL_LEN {
203            match self.reader.read(&mut buf[read_len..]) {
204                Ok(0) if read_len == 0 => return None,
205                Ok(0) => {
206                    return Some(Err(Error::IncompleteRecord {
207                        expected: T::TOTAL_LEN,
208                        actual: read_len,
209                    }));
210                }
211                Ok(n) => read_len += n,
212                Err(err) => return Some(Err(Error::Io(err))),
213            }
214        }
215
216        let record = match T::parse(&buf) {
217            Ok(record) => record,
218            Err(err) => return Some(Err(err)),
219        };
220
221        let separator = self.separator.as_bytes();
222        if !separator.is_empty() {
223            let available = match self.reader.fill_buf() {
224                Ok(bytes) => bytes,
225                Err(err) => return Some(Err(Error::Io(err))),
226            };
227
228            if available.starts_with(separator) {
229                self.reader.consume(separator.len());
230            } else {
231                let actual_len = available.len().min(separator.len());
232                return Some(Err(Error::UnexpectedSeparator {
233                    expected: separator,
234                    actual: available[..actual_len].to_vec(),
235                }));
236            }
237        }
238
239        if let Err(err) = self.check_sequence(&record) {
240            return Some(Err(err));
241        }
242
243        Some(Ok(record))
244    }
245}

247/// Writer for fixed-width records.
248/// 固定長レコードをストリームへ書き出すライターです。
249///
250/// During writing, NUL bytes (`0x00`) are replaced with spaces (`0x20`) and a separator is appended.
251/// 書き込み時は NUL (`0x00`) をスペース (`0x20`) に置換し、レコード末尾に区切りを付けます。
252///
253/// The inner writer is flushed automatically when `Writer` is dropped, but explicit [`Writer::flush`]
254/// is recommended when callers need to handle I/O errors before the writer goes out of scope.
255/// `Writer` が drop されるときに内部 writer は自動的に flush されます。ただし、I/O エラーを明示的に扱いたい場合は
256/// [`Writer::flush`] を呼び出してください。
257///
258/// # Examples
259///
260/// ```
261/// use fixed_record::prelude::*;
262///
263/// #[fixed_record]
264/// pub struct Payment {
265///     id: Fixed<4>,
266///     amount: Fixed<6>,
267/// }
268///
269/// let first = Payment::builder()
270///     .with_id("A001")
271///     .with_amount_int(1200)
272///     .build();
273/// let second = Payment::builder()
274///     .with_id("A002")
275///     .with_amount_int(3400)
276///     .build();
277///
278/// let mut output = Vec::new();
279/// let mut writer = Writer::new(&mut output)
280///     .with_separator(RecordSeparator::Comma);
281///
282/// writer.write_record(&first).unwrap();
283/// writer.write_record(&second).unwrap();
284/// writer.flush().unwrap();
285/// drop(writer);
286///
287/// let mut expected = Vec::new();
288/// expected.extend_from_slice(&first.to_bytes());
289/// expected.push(b',');
290/// expected.extend_from_slice(&second.to_bytes());
291/// expected.push(b',');
292///
293/// assert_eq!(output, expected);
294/// ```
295pub struct Writer<W: Write> {
296    writer: W,
297    separator: &'static [u8],
298}
299
300impl<W: Write> Writer<W> {
301    /// Creates a writer that uses `\n` as the default separator.
302    /// デフォルト区切り `\n` でライターを作成します。
303    pub fn new(writer: W) -> Self {
304        Self {
305            writer,
306            separator: RecordSeparator::Lf.as_bytes(),
307        }
308    }
309
310    /// Changes the separator used after each record.
311    /// レコードごとの区切りを変更します。
312    pub fn with_separator(mut self, separator: RecordSeparator) -> Self {
313        self.separator = separator.as_bytes();
314        self
315    }
316
317    /// Writes one record.
318    /// レコードを1件書き込みます。
319    pub fn write_record<T: FixedRecord>(&mut self, record: &T) -> io::Result<()> {
320        let mut bytes = record.to_bytes();
321        for byte in &mut bytes {
322            if *byte == 0x00 {
323                *byte = b' ';
324            }
325        }
326
327        self.writer.write_all(&bytes)?;
328        self.writer.write_all(self.separator)?;
329        Ok(())
330    }
331
332    /// Flushes the inner writer.
333    /// 内部ライターを flush します。
334    pub fn flush(&mut self) -> io::Result<()> {
335        self.writer.flush()
336    }
337}
338
339impl<W: Write> Drop for Writer<W> {
340    /// Flushes the inner writer when this writer goes out of scope.
341    /// スコープを抜けるときに内部 writer を flush します。
342    fn drop(&mut self) {
343        let _ = self.writer.flush();
344    }
345}

434    impl Write for FlushTrackingWriter {
435        /// Appends bytes to the shared buffer.
436        /// 共有バッファにバイト列を追加します。
437        fn write(&mut self, buf: &[u8]) -> io::Result<usize> {
438            self.bytes.borrow_mut().extend_from_slice(buf);
439            Ok(buf.len())
440        }
441
442        /// Records that a flush was requested.
443        /// flush が要求されたことを記録します。
444        fn flush(&mut self) -> io::Result<()> {
445            *self.flush_count.borrow_mut() += 1;
446            Ok(())
447        }

450    impl BufRead for TrackingBufRead {
451        /// Returns the remaining bytes from the current cursor position.
452        /// 現在のカーソル位置から残りのバイト列を返します。
453        fn fill_buf(&mut self) -> io::Result<&[u8]> {
454            let position = self.cursor.position() as usize;
455            Ok(&self.cursor.get_ref()[position..])
456        }
457
458        /// Records the consumed byte count and advances the cursor.
459        /// 消費したバイト数を記録してカーソルを進めます。
460        fn consume(&mut self, amt: usize) {
461            self.consume_amounts.borrow_mut().push(amt);
462            let position = self.cursor.position() + amt as u64;
463            self.cursor.set_position(position);
464        }
465    }
