3/// Errors that can occur while parsing or operating on fixed-width records.
4/// 固定長データのパース処理や操作中に発生するエラーを表します。
5#[derive(Debug)]
6pub enum Error {
7    /// The input is shorter than the expected fixed width.
8    /// 入力データの長さが、期待される固定長サイズよりも短い場合です。
9    ///
10    /// This can happen when a 10-byte field receives only 5 bytes of data.
11    /// 例えば、10バイト必要なフィールドに対して、5バイトのデータしか与えられなかった場合などに発生します。
12    TooShort,
13
14    /// UTF-8 conversion failed for a byte field.
15    /// バイト列から文字列（UTF-8）への変換に失敗した場合です。
16    ///
17    /// This occurs when a fixed-width field contains invalid UTF-8 bytes.
18    /// 固定長フィールド内に、UTF-8 として不正なバイトシーケンスが含まれている場合に発生します。
19    ///
20    /// For encodings such as Shift_JIS, read the field as bytes and decode it separately.
21    /// Shift_JIS など他のエンコーディングを扱う場合は、バイト列として取得し別途デコードしてください。
22    Utf8Error,
23    /// The input ended in the middle of a fixed-width record.
24    /// 固定長レコードの途中で入力が終わった場合です。
25    IncompleteRecord {
26        expected: usize,
27        actual: usize,
28    },
29    /// An I/O error occurred while using `Reader`, `Writer`, or related operations.
30    /// `Reader` / `Writer` などの I/O 処理中に発生したエラーです。
31    Io(io::Error),
32    /// A field value exceeded the fixed width of the target field.
33    /// フィールドに入れる値が固定長幅を超えた場合です。
34    FieldOverflow {
35        field: &'static str,
36        size: usize,
37        actual: usize,
38    },
39    /// A `Reader` sequence check found a descending key or a forbidden duplicate key.
40    /// `Reader` のシーケンスチェックで、現在レコードが前回レコードより小さい、または同一禁止設定で同一だった場合です。
41    SequenceError {
42        fields: Vec<&'static str>,
43        previous: Vec<Vec<u8>>,
44        current: Vec<Vec<u8>>,
45    },
46    AlignmentError,
47    ParseError,
48}
49
50impl fmt::Display for Error {
51    /// Formats the error as a short user-facing message.
52    /// エラー内容を利用者向けの短いメッセージとして整形します。
53    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
54        match self {
55            Error::TooShort => write!(f, "input data is too short for the fixed length"),
56            Error::Utf8Error => write!(f, "input data contains invalid UTF-8 sequence"),
57            Error::IncompleteRecord { expected, actual } => write!(
58                f,
59                "incomplete fixed record: expected {expected} bytes, got {actual} bytes"
60            ),
61            Error::Io(err) => write!(f, "I/O error while processing fixed record: {err}"),
62            Error::FieldOverflow {
63                field,
64                size,
65                actual,
66            } => write!(
67                f,
68                "field `{field}` is too wide for the fixed length: expected at most {size} bytes, got {actual} bytes"
69            ),
70            Error::SequenceError {
71                fields,
72                previous,
73                current,
74            } => write!(
75                f,
76                "record sequence is not ascending for fields {fields:?}: previous={previous:?}, current={current:?}"
77            ),
78            Error::AlignmentError => write!(f, "Alignment error"),
79            Error::ParseError => write!(f, "failed to parse field value"),
80        }
81    }
82}

84impl PartialEq for Error {
85    /// Compares I/O errors by `ErrorKind` and other errors by variant and payload.
86    /// I/O エラーは `ErrorKind` で比較し、それ以外は variant と値で比較します。
87    fn eq(&self, other: &Self) -> bool {
88        match (self, other) {
89            (Error::TooShort, Error::TooShort) => true,
90            (Error::Utf8Error, Error::Utf8Error) => true,
91            (
92                Error::IncompleteRecord {
93                    expected: left_expected,
94                    actual: left_actual,
95                },
96                Error::IncompleteRecord {
97                    expected: right_expected,
98                    actual: right_actual,
99                },
100            ) => left_expected == right_expected && left_actual == right_actual,
101            (Error::Io(left), Error::Io(right)) => left.kind() == right.kind(),
102            (
103                Error::FieldOverflow {
104                    field: left_field,
105                    size: left_size,
106                    actual: left_actual,
107                },
108                Error::FieldOverflow {
109                    field: right_field,
110                    size: right_size,
111                    actual: right_actual,
112                },
113            ) => {
114                left_field == right_field && left_size == right_size && left_actual == right_actual
115            }
116            (
117                Error::SequenceError {
118                    fields: left_fields,
119                    previous: left_previous,
120                    current: left_current,
121                },
122                Error::SequenceError {
123                    fields: right_fields,
124                    previous: right_previous,
125                    current: right_current,
126                },
127            ) => {
128                left_fields == right_fields
129                    && left_previous == right_previous
130                    && left_current == right_current
131            }
132            (Error::AlignmentError, Error::AlignmentError) => true,
133            (Error::ParseError, Error::ParseError) => true,
134            _ => false,
135        }
136    }
137}

141impl std::error::Error for Error {
142    /// Returns the wrapped I/O error as the source when one exists.
143    /// ラップしている I/O エラーがあれば原因として返します。
144    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
145        match self {
146            Error::Io(err) => Some(err),
147            _ => None,
148        }
149    }
150}

152impl From<io::Error> for Error {
153    /// Converts a standard I/O error into a fixed-record error.
154    /// 標準 I/O エラーを固定長レコード用エラーへ変換します。
