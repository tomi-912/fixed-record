4/// A fixed-width byte buffer whose size `N` is known at compile time.
5/// コンパイル時にサイズ `N` が決定される固定長バイトバッファです。
6///
7/// Internally this stores `[u8; N]`, so values can be handled efficiently on the stack.
8/// 内部的には `[u8; N]` を保持しており、スタック上で効率的に処理されます。
9///
10/// It supports both string-oriented and byte-oriented operations.
11/// 文字列としての操作と、バイト列としての操作の両方をサポートします。
12///
13/// # Generics
14/// * `N`: The byte length of the buffer as a compile-time constant.
15/// * `N`: バイト配列の長さを表すコンパイル時定数です。
16///
17/// # Examples
18///
19/// ```
20/// use fixed_record::Fixed;
21///
22/// // Create a 10-byte buffer and write a string into it.
23/// // 10バイトのバッファを作成し、文字列を書き込みます。
24/// let mut name = Fixed::<10>::spaced();
25/// name.write_bytes(b"Rust");
26///
27/// assert_eq!(name.as_bytes(), b"Rust      ");
28/// assert_eq!(name.as_str().unwrap(), "Rust      ");
29/// ```
30#[repr(C)]
31#[derive(
32    Clone,
33    Copy,
34    PartialEq,
35    Eq,
36    PartialOrd,
37    Ord,
38    Hash,
39    zerocopy::FromBytes,
40    zerocopy::IntoBytes,
41    zerocopy::Immutable,
42    zerocopy::KnownLayout,
43)]
44pub struct Fixed<const N: usize> {
45    /// The fixed-size array that stores the raw bytes.
46    /// 内部データを保持する固定長配列です。
47    pub(crate) buf: [u8; N],
48}
49
50impl<const N: usize> Fixed<N> {
51    /// Copies the first `N` bytes from a byte slice into a new value.
52    /// バイトスライスの先頭から `N` バイトをコピーして、新しいインスタンスを生成します。
53    ///
54    /// # Arguments
55    /// * `src` - The source byte slice to copy from.
56    /// * `src` - コピー元のバイトスライスです。
57    ///
58    /// # Errors
59    /// Returns [`Error::TooShort`] when `src` is shorter than `N` bytes.
60    /// `src` の長さが `N` 未満の場合、[`Error::TooShort`] を返します。
61    ///
62    /// # Examples
63    ///
64    /// ```
65    /// use fixed_record::{Fixed, Error};
66    ///
67    /// // Success case.
68    /// // 正常系です。
69    /// let f = Fixed::<4>::from_slice(b"12345").unwrap();
70    /// assert_eq!(f.as_bytes(), b"1234");
71    ///
72    /// // Error case: the input is too short.
73    /// // 異常系: 入力が短すぎます。
74    /// let err = Fixed::<4>::from_slice(b"123");
75    /// assert!(matches!(err, Err(Error::TooShort)));
76    /// ```
77    pub fn from_slice(src: &[u8]) -> Result<Self, Error> {
78        if src.len() < N {
79            return Err(Error::TooShort);
80        }
81        let mut buf = [0u8; N];
82        buf.copy_from_slice(&src[..N]);
83        Ok(Self { buf })
84    }
85
86    /// Reads `N` bytes from `src` starting at `offset`.
87    /// バイトスライスの指定された位置 `offset` から `N` バイトを読み取ります。
88    ///
89    /// # Arguments
90    /// * `src` - The source byte slice to read from.
91    /// * `src` - 読み取り元のバイトスライスです。
92    /// * `offset` - The byte index where reading starts.
93    /// * `offset` - 読み取り開始位置のバイトインデックスです。
94    ///
95    /// # Errors
96    /// Returns [`Error::TooShort`] when `N` bytes are not available from `offset`.
97    /// 指定された `offset` から `N` バイト分のデータが存在しない場合、[`Error::TooShort`] を返します。
98    ///
99    /// # Examples
100    ///
101    /// ```
102    /// use fixed_record::Fixed;
103    ///
104    /// let raw_data = b"ID001NAME_YAMADA    ";
105    /// // Extract 12 bytes starting at byte 5.
106    /// // 5バイト目から12バイト分を抽出します。
107    /// let name = Fixed::<12>::from_slice_at(raw_data, 5).unwrap();
108    /// assert_eq!(name.as_str().unwrap(), "NAME_YAMADA ");
109    /// ```
110    pub fn from_slice_at(src: &[u8], offset: usize) -> Result<Self, Error> {
111        let end = offset + N;
112        if src.len() < end {
113            return Err(Error::TooShort);
114        }
115        let mut buf = [0u8; N];
116        buf.copy_from_slice(&src[offset..end]);
117        Ok(Self { buf })
118    }
119
120    /// Borrows the stored bytes as a UTF-8 string.
121    /// 保持しているバイト列を UTF-8 文字列として参照します。
122    ///
123    /// # Returns
124    /// A string slice when the stored bytes are valid UTF-8.
125    /// 有効な UTF-8 文字列のスライスです。
126    ///
127    /// # Errors
128    /// Returns [`Error::Utf8Error`] when the stored bytes are not valid UTF-8.
129    /// 内部データが有効な UTF-8 でない場合、[`Error::Utf8Error`] を返します。
130    ///
131    /// # Examples
132    ///
133    /// ```
134    /// use fixed_record::Fixed;
135    ///
136    /// let f = Fixed::from(*b"Hello ");
137    /// assert_eq!(f.as_str().unwrap(), "Hello ");
138    ///
139    /// // Invalid UTF-8 bytes fail.
140    /// // 不正な UTF-8 バイト列はエラーになります。
141    /// let f_bad = Fixed::from([0xFF, 0xFF]);
142    /// assert!(f_bad.as_str().is_err());
143    /// ```
144    pub fn as_str(&self) -> Result<&str, Error> {
145        std::str::from_utf8(&self.buf).map_err(|_| Error::Utf8Error)
146    }
147
148    /// Returns the internal buffer as a byte slice.
149    /// 内部バッファをバイトスライスとして取得します。
150    ///
151    /// This is useful when callers need a zero-copy view of the data.
152    /// ゼロコピーでデータを参照したい場合に便利です。
153    #[inline]
154    pub fn as_bytes(&self) -> &[u8] {
155        &self.buf
156    }
157
158    /// Creates a value filled with `0x00` bytes.
159    /// すべてのバイトが `0x00` で埋められたインスタンスを生成します。
160    ///
161    /// # Examples
162    /// ```
163    /// use fixed_record::Fixed;
164    /// let f: Fixed<4> = Fixed::zeroed();
165    /// assert_eq!(f.as_bytes(), &[0, 0, 0, 0]);
166    /// ```
167    pub const fn zeroed() -> Self {
168        Self { buf: [0u8; N] }
169    }
170
171    /// Creates a value filled with `0x20` space bytes.
172    /// すべてのバイトが `0x20` の半角スペースで埋められたインスタンスを生成します。
173    ///
174    /// This is useful for text fixed-width files that use space padding.
175    /// テキスト形式の固定長ファイルで、空白埋めが必要な場合に便利です。
176    ///
177    /// # Examples
178    /// ```
179    /// use fixed_record::Fixed;
180    /// let f: Fixed<3> = Fixed::spaced();
181    /// assert_eq!(f.as_bytes(), b"   ");
182    /// ```
183    pub const fn spaced() -> Self {
184        Self { buf: [b' '; N] }
185    }
186
187    /// Creates a value filled with the specified byte.
188    /// すべてのバイトが指定された値で埋められたインスタンスを生成します。
189    pub const fn filled(byte: u8) -> Self {
190        Self { buf: [byte; N] }
191    }
192
193    /// Writes arbitrary bytes into the buffer.
194    /// 任意のバイト列をバッファに書き込みます。
195    ///
196    /// # Behavior
197    /// # 挙動
198    /// - If `src` is longer than `N`, only the first `N` bytes are written and the rest is truncated.
199    /// - `src` の長さが `N` より長い場合、先頭 `N` バイトのみを書き込み、残りは切り捨てます。
200    /// - If `src` is shorter than `N`, only the leading bytes are replaced and the remaining bytes keep their previous values.
201    /// - `src` の長さが `N` より短い場合、先頭部分だけを書き換え、残りのバイトは以前の値を維持します。
202    ///
203    /// # Arguments
204    /// * `src` - The bytes to write.
205    /// * `src` - 書き込むバイト列です。
206    ///
207    /// # Examples
208    ///
209    /// ```
210    /// use fixed_record::Fixed;
211    ///
212    /// // Truncation.
213    /// // 切り捨ての例です。
214    /// let mut f = Fixed::<3>::spaced();
215    /// f.write_bytes(b"ABCDEFG");
216    /// assert_eq!(f.as_bytes(), b"ABC");
217    ///
218    /// // Partial overwrite: the remaining spaces are preserved.
219    /// // 部分書き換えの例です。残りのスペースは維持されます。
220    /// let mut f = Fixed::<5>::spaced();
221    /// f.write_bytes(b"Hi");
222    /// assert_eq!(f.as_bytes(), b"Hi   ");
223    /// ```
224    pub fn write_bytes(&mut self, src: &[u8]) {
225        let len = src.len().min(N);
226        self.buf[..len].copy_from_slice(&src[..len]);
227    }
228
229    /// Fills the internal buffer with the specified byte.
230    /// 内部バッファを指定バイトで上書きします。
231    pub fn fill(&mut self, byte: u8) {
232        self.buf = [byte; N];
233    }
234
235    /// Fills the internal buffer with `0x00` bytes.
236    /// 内部バッファをすべて `0x00` で上書きします。
237    pub fn fill_zero(&mut self) {
238        self.buf = [0u8; N];
239    }
240
241    /// Fills the internal buffer with `0x20` space bytes.
242    /// 内部バッファをすべて半角スペース (`0x20`) で上書きします。
243    pub fn fill_space(&mut self) {
244        self.buf = [b' '; N];
245    }
246}

248impl<const N: usize> fmt::Debug for Fixed<N> {
249    /// Formats as a string when the bytes are valid UTF-8, otherwise as a byte array.
250    /// UTF-8 として有効なら文字列形式で、そうでなければバイト配列形式で出力します。
251    ///
252    /// # Examples
253    /// ```
254    /// use fixed_record::Fixed;
255    /// let f = Fixed::from(*b"ABC");
256    /// assert_eq!(format!("{:?}", f), "Fixed(\"ABC\")");
257    /// ```
258    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
259        match self.as_str() {
260            Ok(s) => write!(f, "Fixed(\"{}\")", s),
261            Err(_) => write!(f, "Fixed({:?})", &self.buf),
262        }
263    }
264}

266impl<const N: usize> fmt::Display for Fixed<N> {
267    /// Displays the internal bytes directly as a string.
268    /// 内部のバイト列を直接文字列として出力します。
269    ///
270    /// Invalid UTF-8 is displayed as `<?>`.
271    /// 不正な UTF-8 が含まれる場合は `<?>` を表示します。
272    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
273        let s = std::str::from_utf8(&self.buf).unwrap_or("<?>");
274        write!(f, "{}", s)
275    }
276}
